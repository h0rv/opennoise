def metadata_scores(q):
    result=np.zeros(N)
    for col,value in zip(X.indices[X.indptr[q]:X.indptr[q+1]],X.data[X.indptr[q]:X.indptr[q+1]],strict=True):

def rank_metrics(score,q,targets):
    candidate=np.flatnonzero((score>threshold)&(np.arange(N)!=q));rounded=np.round(score[candidate],8)
    ranks=[]
    for p in targets:
        if score[p]<=threshold:continue
        value=np.round(score[p],8)
        ranks.append(1+int(np.count_nonzero(rounded>value))+int(np.count_nonzero((rounded==value)&(candidate<p))))
    mrr=1/min(ranks)if ranks else 0.
    if len(candidate)>10:
        boundary=np.partition(rounded,len(rounded)-10)[len(rounded)-10];short=candidate[rounded>=boundary]
    else:short=candidate
    top=short[np.lexsort((short,-np.round(score[short],8)))][:10]
    return {'Recall10':sum(r<=10 for r in ranks)/len(targets),'Recall50':sum(r<=50 for r in ranks)/len(targets),'MRR':mrr,'returned_positive_candidates':len(candidate),'top10':[ids[i]for i in top]}, {'MRR50':mrr if ranks and min(ranks)<=50 else 0.,'returned_at50':min(50,len(candidate)),'has_positive_candidates':bool(len(candidate))}

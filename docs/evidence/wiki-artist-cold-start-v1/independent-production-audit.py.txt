def sha(p):
    with p.open('rb')as f:return hashlib.file_digest(f,'sha256').hexdigest()

files=[SOURCE,PUBLIC,OLDROOT/'data/examples/wiki-artist-atlas-v1/snapshot.json.gz',ML/'.cache/wiki-artist-model-v5/metadata.npz',OUT.parent/'wiki-artist-cold-start-v1-independent-review/report.json',ROOT/'src/opennoise/pipeline/wiki_artist_atlas.py',SITE/'index.html',SITE/'atlas.mjs',SITE/'atlas.css',SITE/'atlas/receipt.json',OLD/'opennoise-release.json']

source=json.load(gzip.open(SOURCE,'rt'));public=json.load(gzip.open(PUBLIC,'rt'))

assert public['revision']=='wiki-artist-atlas-v1'
assert set(public['source'])==set(source['source'])|{'navigation_layout'}
assert public['source']['navigation_layout']
del public['source']['navigation_layout']
public['revision']=source['revision']
assert public==source,'Native source facts changed'
del public
artists=source['artists'];uris=[a['uri']for a in artists];N=len(artists)
assert N==86253 and uris==sorted(set(uris))
index={u:i for i,u in enumerate(uris)}
public_ids=[hashlib.sha256(u.encode()).hexdigest()[:32]for u in uris]

public_index={u:i for i,u in enumerate(public_ids)}
neighbors=[set()for _ in uris]
for a,_,b in source['relations']:
    i,j=index[a],index[b];neighbors[i].add(j);neighbors[j].add(i)

assert len(isolated)==30813 and np.count_nonzero(degree)==55440
details={};old_details={}

    for p in sorted(directory.glob('*.json')):
        part=json.loads(p.read_bytes());assert not set(target)&set(part);target.update(part)

assert len(prior_json)==961

    else:

        assert not row['related']and not row['map']and not old['suggested']

# Reconstruct benchmark features directly from accepted native facts and document frequencies.
canonical={g['uri']:('core:'+g['vocabulary_term']if g['vocabulary_term']else'typed:'+unicodedata.normalize('NFKC',g['name']).casefold().strip())for g in source['genres']}
rr=[];cc=[];vv=[];offset=0;block_count=np.zeros(N,dtype=np.int8);blocks={}
for kind in ('genre','instrument','recordLabel'):
    membership=[sorted({canonical[g]for g in a['genres']})if kind=='genre'else sorted(set(a['features'].get('http://dbpedia.org/ontology/'+kind,[])))for a in artists]
    frequency=Counter(v for row in membership for v in row);vocab=sorted(frequency);columns={v:i+offset for i,v in enumerate(vocab)}
    weight={v:1+math.log((N+1)/(frequency[v]+1))for v in vocab}
    for i,row in enumerate(membership):
        if not row:continue
        block_count[i]+=1;length=math.sqrt(sum(weight[v]**2 for v in row))
        for v in row:rr.append(i);cc.append(columns[v]);vv.append(weight[v]/length)
    offset+=len(vocab);blocks[kind]={'features':len(vocab),'artists':sum(bool(row)for row in membership)}
rr=np.asarray(rr,dtype=np.int32);vv=np.asarray(vv)/np.sqrt(block_count[rr].astype(float))
X=sparse.coo_matrix((vv,(rr,np.asarray(cc,dtype=np.int32))),shape=(N,offset)).tocsr();X.sort_indices()
benchmark=sparse.load_npz(ML/'.cache/wiki-artist-model-v5/metadata.npz').tocsr();delta=X-benchmark
matrix_error=float(np.max(np.abs(delta.data),initial=0));assert X.shape==benchmark.shape and matrix_error<2e-14
assert (np.diff(X.indptr)[isolated]>0).all()
XC=X.tocsc();threshold=1e-8
del benchmark,delta,rr,cc,vv,membership,weight,frequency,columns
progress('metadata_independently_reconstructed',shape=list(X.shape),maximum_error=matrix_error)

proofs=[];abstentions=[];qualifying=0;minimum_score=1.;source_labels={a['id']:a['name']for a in json.loads((SITE/'atlas/artists.json').read_bytes())}
for step,q in enumerate(isolated):
    aid=public_ids[q];row=details[aid];chosen=row['suggested']
    assert len(chosen)==len(set(chosen))<=10 and all(i in public_index for i in chosen)and aid not in chosen
    assert not neighbors[q]
    score=np.zeros(N)
    for feature,value in zip(X.indices[X.indptr[q]:X.indptr[q+1]],X.data[X.indptr[q]:X.indptr[q+1]],strict=True):
        low,high=XC.indptr[feature:feature+2];np.add.at(score,XC.indices[low:high],value*XC.data[low:high])
    score[q]=0.;candidates=np.flatnonzero(score>threshold)
    assert len(chosen)==min(10,len(candidates)),('Output count differs',uris[q],len(chosen),len(candidates))
    if not chosen:
        assert not candidates.size and 'suggestion_basis'not in row
        abstentions.append({'uri':uris[q],'id':aid,'name':row['name'],'metadata_nonempty':True,'positive_candidates':0});continue
    assert row.get('suggestion_basis')=='artist_metadata'
    chosen_indices=[public_index[i]for i in chosen]
    assert (np.diff(X.indptr)[chosen_indices]>0).all()
    assert all(score[i]>threshold and source_labels[public_ids[i]].strip()for i in chosen_indices)
    keys=[(-float(np.round(score[i],8)),i)for i in chosen_indices]
    assert all(a<b for a,b in zip(keys,keys[1:])),('Incorrect stable order',uris[q])
    # Rank-boundary proof: sorted k positive distinct outputs whose last global rank is k
    # must be the complete correct top-k. No producer sort/selection implementation reused.
    last=chosen_indices[-1];last_value=np.round(score[last],8);rounded=np.round(score[candidates],8)
    last_rank=1+int(np.count_nonzero(rounded>last_value))+int(np.count_nonzero((rounded==last_value)&(candidates<last)))
    assert last_rank==len(chosen),('Top-k boundary differs',uris[q],last_rank,len(chosen))
    qualifying+=1;minimum_score=min(minimum_score,min(float(score[i])for i in chosen_indices))
    proofs.append({'uri':uris[q],'id':aid,'name':row['name'],'positive_candidates':len(candidates),'suggested':chosen,'last_global_rank':last_rank,'boundary_score8':float(last_value)})
    if step%2500==0:progress('ranking_boundary_proof',checked=step+1,total=len(isolated),nonempty=qualifying,abstentions=len(abstentions))

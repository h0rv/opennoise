def sha(path):
    h=hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda:stream.read(1_048_576),b''):h.update(block)
    return h.hexdigest()

def inverted_cosine(matrix,csc,q):
    scores=np.zeros(N,dtype=np.float64)
    for feature,weight in zip(matrix.indices[matrix.indptr[q]:matrix.indptr[q+1]],matrix.data[matrix.indptr[q]:matrix.indptr[q+1]],strict=True):
        lo,hi=csc.indptr[feature:feature+2]
        np.add.at(scores,csc.indices[lo:hi],csc.data[lo:hi]*weight)
    return scores

def positive_metrics(scores,candidates,targets,total):
    # Count preceding candidates for each positive directly; no producer sorting/ranking code.
    rounded=np.round(scores[candidates],8)
    ranks=[]
    for p in targets:
        score=np.round(scores[p],8)
        ranks.append(1+int(np.count_nonzero(rounded>score))+int(np.count_nonzero((rounded==score)&(candidates<p))))
    return {'Recall10':sum(r<=10 for r in ranks)/total,'Recall50':sum(r<=50 for r in ranks)/total,'MRR':1/min(ranks) if ranks else 0.0}

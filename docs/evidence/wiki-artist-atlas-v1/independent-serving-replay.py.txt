def positive_metrics(scores,candidates,targets):
    # Different ranking calculation: count preceding candidates, never sort all scores.
    rounded=np.round(scores[candidates],8)
    ranks=[]
    for p in targets:
        value=np.round(scores[p],8)
        ranks.append(1+int(np.count_nonzero(rounded>value))+int(np.count_nonzero((rounded==value)&(candidates<p))))
    return {'Recall10':sum(r<=10 for r in ranks)/len(targets),'Recall50':sum(r<=50 for r in ranks)/len(targets),'MRR':1/min(ranks)}
